**Program.cs**

using OpenAI.Chat;

using System.Text.Json;

using System.Text.Json.Serialization;

ChatClient client = new ChatClient(

model: "gpt-4o",

apiKey: "sk-proj-M3yaB9HwobWDHJ1hp\_BTOmO\_FUrAA7KKZZZA5n\_H29DBfQu3-hCWjJ43KkTF8CvtQ9I-9vM0ryT3BlbkFJ1Z\_7yq9TSkj-JKxlAdd3PvrecCAT2\_mmztLhxTvyUPKg2WvA1bXqnh1B1DouzGUM9ICJ620a8A"

);

List<ChatMessage> messages = new List<ChatMessage>

{

new SystemChatMessage("""

You are a poet.

You can write poems with variable number of lines in each stanza.

Please use the below json format for output

{

"stanza": [

{

"lines: [

"Line 1"

],

"lines\_count": 3

}

]

}

"""),

};

messages.Add(new UserChatMessage("Animals"));

ChatCompletionOptions options = new ChatCompletionOptions();

options.ResponseFormat = ChatResponseFormat.CreateJsonObjectFormat();

ChatCompletion completion = client.CompleteChat(messages, options);

Console.WriteLine(completion.Content[0].Text);

Poem? poem = JsonSerializer.Deserialize<Poem>(completion.Content[0].Text);

if (poem == null)

{

Console.WriteLine("Poem is null");

return;

}

foreach (var stanza in poem.Stanza)

{

Console.WriteLine("Stanza:");

foreach (var line in stanza.Lines)

{

Console.WriteLine($" {line}");

}

Console.WriteLine($"Line Count: {stanza.LinesCount}\n");

}

using OpenAI.Chat;

using System.ClientModel;

using System.Data;

ChatClient client = new ChatClient(

model: "gpt-4o",

apiKey: "sk-proj-M3yaB9HwobWDHJ1hp\_BTOmO\_FUrAA7KKZZZA5n\_H29DBfQu3-hCWjJ43KkTF8CvtQ9I-9vM0ryT3BlbkFJ1Z\_7yq9TSkj-JKxlAdd3PvrecCAT2\_mmztLhxTvyUPKg2WvA1bXqnh1B1DouzGUM9ICJ620a8A"

);

List<ChatMessage> messages = new List<ChatMessage>

{

new SystemChatMessage("\"You are a professor of AI and can answer all questions related to AI and ML for students of class 10"),

};

messages.Add(new UserChatMessage("What is AI?"));

ChatCompletionOptions options = new ChatCompletionOptions();

options.Temperature = 0.2F;

options.PresencePenalty = 0.0F;

CollectionResult<StreamingChatCompletionUpdate> completionUpdates = client.CompleteChatStreaming(messages, options);

foreach (StreamingChatCompletionUpdate completionUpdate in completionUpdates)

{

if (completionUpdate.ContentUpdate.Count > 0)

{

Console.Write(completionUpdate.ContentUpdate[0].Text);

Console.WriteLine("");

}

}
